@startuml
class Cliente {
  +id: Long
  +nombreCompleto: String
  +fechaCreacion: LocalDateTime

  --
  +documento: DocumentoIdentificativo

class SolicitudPrestamo {
  +id: Long
  +importe: BigDecimal
  +estado: EstadoSolicitud

  +fechaModificacion: LocalDateTime
}

class DocumentoIdentificativo {
  +tipo: TipoDocumento
  +numero: String
}

  +id: Long
  +codigo: String
  +descripcion: String
}

enum TipoDocumento {
  DNI
  NIE
  PASSPORT
}

enum EstadoSolicitud {
  PENDIENTE
  APROBADA
  RECHAZADA
  CANCELADA
}

SolicitudPrestamo --> EstadoSolicitud
DocumentoIdentificativo --> TipoDocumento
